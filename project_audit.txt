import time
import cv2
import torch
from ultralytics import YOLO

MODELS = {
    "YOLO11n": r"D:\person-reid\yolo11n.pt",
    "YOLO26s": r"D:\person-reid\yolo26s.pt",
}

IMAGE_DIR = r"D:\person-reid\data\evaluation"
IMAGE_SIZE = 416
CONFIDENCE = 0.35
WARMUP_RUNS = 3
BENCHMARK_RUNS = 10


def main():
    print("Device: CPU")
    print(f"PyTorch: {torch.__version__}")
    print(f"OpenCV: {cv2.__version__}")
    print(f"Image size: {IMAGE_SIZE}")
    print()

    # Use the same image set for both models.
    import os

    images = []
    for root, _, files in os.walk(IMAGE_DIR):
        for filename in files:
            path = os.path.join(root, filename)
            image = cv2.imread(path)
            if image is not None:
                images.append((path, image))

    if not images:
        print(f"No readable images found in {IMAGE_DIR}")
        return

    print(f"Images found: {len(images)}")
    print()

    results_summary = []

    for model_name, model_path in MODELS.items():
        print("=" * 60)
        print(f"Testing {model_name}: {model_path}")

        model = YOLO(model_path)

        # Warm up the model.
        for i in range(min(WARMUP_RUNS, len(images))):
            model.predict(
                source=images[i][1],
                imgsz=IMAGE_SIZE,
                conf=CONFIDENCE,
                classes=[0],
                device="cpu",
                verbose=False,
            )

        total_time = 0.0
        total_people = 0
        total_confidence = 0.0
        confidence_count = 0
        timings = []

        for i in range(BENCHMARK_RUNS):
            image_path, image = images[i % len(images)]

            start = time.perf_counter()

            prediction = model.predict(
                source=image,
                imgsz=IMAGE_SIZE,
                conf=CONFIDENCE,
                classes=[0],
                device="cpu",
                verbose=False,
            )

            elapsed = time.perf_counter() - start
            timings.append(elapsed)
            total_time += elapsed

            boxes = prediction[0].boxes
            count = len(boxes) if boxes is not None else 0
            total_people += count

            if boxes is not None and count:
                scores = boxes.conf.cpu().tolist()
                total_confidence += sum(scores)
                confidence_count += len(scores)

            print(
                f"Run {i + 1:02d}: "
                f"{elapsed * 1000:.1f} ms, "
                f"people detected: {count}, "
                f"image: {os.path.basename(image_path)}"
            )

        average_ms = (total_time / BENCHMARK_RUNS) * 1000
        average_fps = 1.0 / (total_time / BENCHMARK_RUNS)
        average_confidence = (
            total_confidence / confidence_count
            if confidence_count else 0.0
        )

        results_summary.append(
            (
                model_name,
                average_ms,
                average_fps,
                total_people / BENCHMARK_RUNS,
                average_confidence,
            )
        )

        print(f"\n{model_name} results:")
        print(f"Average inference time: {average_ms:.1f} ms")
        print(f"Approximate inference FPS: {average_fps:.2f}")
        print(f"Average people per image: {total_people / BENCHMARK_RUNS:.2f}")
        print(f"Mean confidence: {average_confidence:.3f}")
        print()

        del model

    print("\n" + "=" * 75)
    print("FINAL COMPARISON")
    print("-" * 75)
    print(
        f"{'Model':<12} {'Avg ms':>10} {'FPS':>10} "
        f"{'People/image':>15} {'Mean conf.':>12}"
    )

    for name, ms, fps, people, confidence in results_summary:
        print(
            f"{name:<12} {ms:>10.1f} {fps:>10.2f} "
            f"{people:>15.2f} {confidence:>12.3f}"
        )

    print("\nNote: this measures detector inference on still images,")
    print("not end-to-end RTSP tracking or cross-camera Re-ID.")


if __name__ == "__main__":
    main()

from pathlib import Path
import sys
import csv

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from src.recognition.face_recognizer import FaceRecognizer


GALLERY_PATH = ROOT / "data" / "staff_gallery"
EVALUATION_PATH = ROOT / "data" / "evaluation"
OUTPUT_PATH = ROOT / "tests" / "face_evaluation_results.csv"

SUPPORTED_EXTENSIONS = {
    ".jpg", ".jpeg", ".png", ".bmp"
}


def get_evaluation_images():
    images = []

    if not EVALUATION_PATH.exists():
        raise FileNotFoundError(
            f"Evaluation directory not found: {EVALUATION_PATH}"
        )

    for person_dir in sorted(EVALUATION_PATH.iterdir()):
        if not person_dir.is_dir():
            continue

        expected_name = person_dir.name

        for image_path in sorted(person_dir.iterdir()):
            if image_path.suffix.lower() not in SUPPORTED_EXTENSIONS:
                continue

            images.append((image_path, expected_name))

    return images


def evaluate():
    recognizer = FaceRecognizer()

    print("Loading enrolled staff gallery...")
    gallery = recognizer.load_gallery(GALLERY_PATH)

    print("\nEnrolled identities:")
    for name, embeddings in gallery.items():
        print(f"  {name}: {len(embeddings)} reference image(s)")

    evaluation_images = get_evaluation_images()

    if not evaluation_images:
        raise RuntimeError(
            f"No evaluation images found in {EVALUATION_PATH}"
        )

    rows = []

    for image_path, expected_name in evaluation_images:
        print(f"\nEvaluating: {expected_name}/{image_path.name}")

        try:
            query_embedding = recognizer.extract_embedding(image_path)
            ranked = recognizer.rank_candidates(query_embedding, gallery)

            best_match = ranked[0]
            best_name = best_match["name"]
            best_score = best_match["similarity"]

            correct_top1 = (
                expected_name != "Unknown"
                and best_name == expected_name
            )

            row = {
                "image": str(image_path.relative_to(ROOT)),
                "expected_identity": expected_name,
                "best_candidate": best_name,
                "best_similarity": round(best_score, 6),
                "correct_top1": (
                    correct_top1 if expected_name != "Unknown" else ""
                ),
                "top2_candidate": (
                    ranked[1]["name"] if len(ranked) > 1 else ""
                ),
                "top2_similarity": (
                    round(ranked[1]["similarity"], 6)
                    if len(ranked) > 1 else ""
                ),
                "status": "processed",
            }

            for candidate in ranked:
                print(
                    f"  {candidate['name']}: "
                    f"{candidate['similarity']:.4f}"
                )

            print(f"  Best candidate: {best_name}")
            print(f"  Best similarity: {best_score:.4f}")

            if expected_name == "Unknown":
                print(
                    "  NOTE: This is an unknown-person test. "
                    "A high-scoring staff candidate may be a false match."
                )
            else:
                print(f"  Expected identity: {expected_name}")
                print(f"  Correct top-1 candidate: {correct_top1}")

        except ValueError as exc:
            print(f"  SKIPPED: {exc}")

            row = {
                "image": str(image_path.relative_to(ROOT)),
                "expected_identity": expected_name,
                "best_candidate": "",
                "best_similarity": "",
                "correct_top1": "",
                "top2_candidate": "",
                "top2_similarity": "",
                "status": f"error: {exc}",
            }

        rows.append(row)

    fieldnames = [
        "image",
        "expected_identity",
        "best_candidate",
        "best_similarity",
        "correct_top1",
        "top2_candidate",
        "top2_similarity",
        "status",
    ]

    with OUTPUT_PATH.open(
        "w",
        newline="",
        encoding="utf-8",
    ) as file:
        writer = csv.DictWriter(file, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)

    known_rows = [
        row for row in rows
        if row["expected_identity"] != "Unknown"
        and row["status"] == "processed"
    ]

    unknown_rows = [
        row for row in rows
        if row["expected_identity"] == "Unknown"
        and row["status"] == "processed"
    ]

    print("\n" + "=" * 55)
    print("EVALUATION SUMMARY")
    print("=" * 55)

    if known_rows:
        correct = sum(
            row["correct_top1"] is True
            for row in known_rows
        )

        print(f"Known-person images processed: {len(known_rows)}")
        print(f"Correct top-1 matches: {correct}/{len(known_rows)}")
        print(
            "Top-1 accuracy on this small test set: "
            f"{correct / len(known_rows):.1%}"
        )
        print(
            "This is a preliminary measurement, not a guarantee "
            "of CCTV performance."
        )
    else:
        print("No known-person evaluation images were processed.")

    if unknown_rows:
        print(f"\nUnknown-person images processed: {len(unknown_rows)}")
        print(
            "Unknown images whose highest-scoring candidate is a staff "
            f"member: {len(unknown_rows)}/{len(unknown_rows)}"
        )
        print(
            "These are candidate rankings, not confirmed identity matches. "
            "We have not applied an acceptance threshold yet."
        )
    else:
        print("\nNo unknown-person evaluation images were processed.")

    print(f"\nDetailed results saved to:\n{OUTPUT_PATH}")


if __name__ == "__main__":
    evaluate()
    
def identify(
    self,
    query_embedding,
    gallery,
    threshold=0.30,
):
    """
    Prototype open-set identification.

    The threshold is configurable and has NOT been validated
    for production use.
    """
    ranked = self.rank_candidates(query_embedding, gallery)

    if not ranked:
        return {
            "identity": "Unknown",
            "similarity": None,
            "status": "no_candidates",
            "candidates": [],
        }

    best = ranked[0]

    if best["similarity"] < threshold:
        return {
            "identity": "Unknown",
            "similarity": best["similarity"],
            "status": "below_threshold",
            "candidates": ranked,
        }

    return {
        "identity": best["name"],
        "similarity": best["similarity"],
        "status": "candidate_match",
        "candidates": ranked,
    }

from pathlib import Path
import sys
import csv

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from src.recognition.face_recognizer import FaceRecognizer

GALLERY_PATH = ROOT / "data" / "staff_gallery"
EVALUATION_PATH = ROOT / "data" / "evaluation"
OUTPUT_PATH = ROOT / "tests" / "threshold_results.csv"

SUPPORTED_EXTENSIONS = {
    ".jpg", ".jpeg", ".png", ".bmp"
}

THRESHOLDS = [
    round(value / 100, 2)
    for value in range(20, 71, 5)
]

# Exact duplicates of enrolled images, excluded from evaluation.
EXCLUDED_IMAGES = {
    Path("Dhruval")
    / "WhatsApp Image 2026-10-12 at 5.58.41 PM.jpeg",

    Path("Prathamesh")
    / "WIN_20261012_17_25_11_Pro.jpg",
}


def main():
    recognizer = FaceRecognizer()
    gallery = recognizer.load_gallery(GALLERY_PATH)

    records = []
    skipped = []

    for person_dir in sorted(EVALUATION_PATH.iterdir()):
        if not person_dir.is_dir():
            continue

        expected_name = person_dir.name

        for image_path in sorted(person_dir.iterdir()):
            if image_path.suffix.lower() not in SUPPORTED_EXTENSIONS:
                continue

            relative_path = image_path.relative_to(EVALUATION_PATH)

            if relative_path in EXCLUDED_IMAGES:
                print(f"Excluded duplicate: {relative_path}")
                continue

            try:
                embedding = recognizer.extract_embedding(image_path)

                ranked = recognizer.rank_candidates(
                    embedding,
                    gallery,
                )

                if not ranked:
                    skipped.append((relative_path, "No candidates"))
                    continue

                records.append({
                    "image": str(relative_path),
                    "expected": expected_name,
                    "best_name": ranked[0]["name"],
                    "similarity": ranked[0]["similarity"],
                })

            except ValueError as exc:
                skipped.append((relative_path, str(exc)))

    if not records:
        print("No usable evaluation images found.")
        return

    output_rows = []

    print("\nTHRESHOLD EVALUATION")
    print("=" * 90)

    print(
        f"{'Threshold':<12}"
        f"{'Known accepted':<18}"
        f"{'Known rejected':<18}"
        f"{'Unknown accepted':<20}"
        f"{'Unknown rejected':<20}"
    )

    for threshold in THRESHOLDS:
        known_accepted = 0
        known_rejected = 0
        unknown_accepted = 0
        unknown_rejected = 0
        known_wrong_identity = 0

        for record in records:
            expected = record["expected"]
            predicted = record["best_name"]
            similarity = record["similarity"]

            accepted = similarity >= threshold

            if expected == "Unknown":
                if accepted:
                    unknown_accepted += 1
                else:
                    unknown_rejected += 1

            else:
                if not accepted:
                    known_rejected += 1
                elif predicted == expected:
                    known_accepted += 1
                else:
                    known_wrong_identity += 1

        row = {
            "threshold": threshold,
            "known_accepted_correct": known_accepted,
            "known_rejected": known_rejected,
            "known_accepted_wrong_identity": known_wrong_identity,
            "unknown_accepted": unknown_accepted,
            "unknown_rejected": unknown_rejected,
            "known_images": sum(
                record["expected"] != "Unknown"
                for record in records
            ),
            "unknown_images": sum(
                record["expected"] == "Unknown"
                for record in records
            ),
        }

        output_rows.append(row)

        print(
            f"{threshold:<12.2f}"
            f"{known_accepted:<18}"
            f"{known_rejected:<18}"
            f"{unknown_accepted:<20}"
            f"{unknown_rejected:<20}"
        )

    with OUTPUT_PATH.open(
        "w",
        newline="",
        encoding="utf-8",
    ) as file:
        writer = csv.DictWriter(
            file,
            fieldnames=output_rows[0].keys(),
        )
        writer.writeheader()
        writer.writerows(output_rows)

    print(f"\nCSV saved to: {OUTPUT_PATH}")

    print("\nSKIPPED IMAGES")
    for image, reason in skipped:
        print(f"{image}: {reason}")

    x1 = max(0, min(x1, width - 1))
    y1 = max(0, min(y1, height - 1))

    y2 = max(0, min(y2, height))
    if x2 <= x1 or y2 <= y1:

def main():

        raise RuntimeError("CAMERA_202_RTSP is not set in this PowerShell session.")

        min_face_width=MIN_FACE_WIDTH,
        min_face_height=MIN_FACE_HEIGHT,
        min_blur_score=MIN_BLUR_SCORE,
        min_detection_confidence=MIN_FACE_DETECTION_CONFIDENCE,
    )

    person_model = YOLO(PERSON_MODEL)

    print("[4/4] Connecting to RTSP stream...")
    params = [
        cv2.CAP_PROP_OPEN_TIMEOUT_MSEC, 10000,
        cv2.CAP_PROP_READ_TIMEOUT_MSEC, 10000,

        cap.release()
        raise RuntimeError("Could not open RTSP stream. Check camera and URL.")

    print("Camera connected. Recognition is running. Keep the RTSP URL private.\n")

    track_identity = {}
    track_last_face_time = {}
    track_last_log_time = {}

    frame_count = 0
    face_pass_count = 0
    start_time = time.perf_counter()
    last_face_pass = 0.0
    last_profile_time = start_time
    last_profile_frame = 0
    last_profile_face_pass = 0

    # Timings accumulate between profile summaries.
    stats = defaultdict(float)
    total_face_calls = 0

    try:
        while True:
            loop_start = time.perf_counter()
            read_start = time.perf_counter()
            ok, frame = cap.read()
            read_time = time.perf_counter() - read_start
            if not ok or frame is None:
                print("Frame read failed. Stopping the pipeline.")
                break

            frame_count += 1
            frame_height, frame_width = frame.shape[:2]
            current_time = time.time()

            yolo_start = time.perf_counter()
            results = person_model.track(
                frame, persist=True, tracker="bytetrack.yaml",
                classes=[0], conf=PERSON_CONFIDENCE,
                imgsz=IMAGE_SIZE, device="cpu", verbose=False,
            )
            yolo_time = time.perf_counter() - yolo_start
            stats["yolo"] += yolo_time
            stats["read"] += read_time

            people = []
            if results and results[0].boxes is not None:
                boxes = results[0].boxes
                if boxes.id is not None:
                    xyxy = boxes.xyxy.cpu().numpy()
                    ids = boxes.id.int().cpu().tolist()
                    confidences = boxes.conf.cpu().tolist()
                    for box, track_id, confidence in zip(xyxy, ids, confidences):
                        person_box = clamp_box(box, frame_width, frame_height)
                        if person_box is not None:
                            people.append({
                                "track_id": int(track_id),
                                "box": person_box,
                                "confidence": float(confidence),
                            })

            face_pass_due = current_time - last_face_pass >= FACE_PROCESS_INTERVAL
            if face_pass_due:
                last_face_pass = current_time
                face_pass_count += 1
                face_pass_start = time.perf_counter()

                for person in people:
                    track_id = person["track_id"]
                    px1, py1, px2, py2 = person["box"]
                    person_width = px2 - px1
                    person_height = py2 - py1
                    if (person_width < MIN_PERSON_CROP_WIDTH or
                            person_height < MIN_PERSON_CROP_HEIGHT):

                    crop_y2 = py1 + int(person_height * PERSON_CROP_BOTTOM_RATIO)
                    person_crop = frame[py1:crop_y2, px1:px2]
                    if person_crop.size == 0:
                        continue

                    face_start = time.perf_counter()
                    faces = face_recognizer.app.get(person_crop)
                    face_elapsed = time.perf_counter() - face_start
                    stats["face_detect"] += face_elapsed
                    stats["face_calls"] += 1
                    total_face_calls += 1

                    # Avoid choosing arbitrarily when the crop has zero or multiple faces.
                    if len(faces) != 1:
                        continue

                    match_start = time.perf_counter()
                    face = faces[0]
                    quality = quality_checker.evaluate(person_crop, face)
                    if not quality.acceptable:
                        stats["match"] += time.perf_counter() - match_start
                        continue

                    track_last_face_time[track_id] = current_time
                    embedding = face_recognizer.normalize_embedding(face.normed_embedding)
                    ranked = face_recognizer.rank_candidates(embedding, gallery)
                    if not ranked:
                        stats["match"] += time.perf_counter() - match_start
                        continue

                    best = ranked[0]
                    best_name = best["name"]
                    best_similarity = float(best["similarity"])
                    second_similarity = (
                        float(ranked[1]["similarity"]) if len(ranked) > 1 else None
                    )
                    margin_ok = (
                        second_similarity is None or
                        best_similarity - second_similarity >= MIN_SIMILARITY_MARGIN
                    )
                    candidate_accepted = (
                        best_similarity >= FACE_SIMILARITY_THRESHOLD and margin_ok
                    )
                    match_history[track_id].append(
                        best_name if candidate_accepted else "Unknown"
                    )

                    votes = list(match_history[track_id])
                    confirmed_name = "Unknown"
                    if votes:
                        most_common_name = max(set(votes), key=votes.count)
                        if (most_common_name != "Unknown" and
                                votes.count(most_common_name) >= MIN_MATCH_VOTES):
                            confirmed_name = most_common_name

                    previous_name = track_identity.get(track_id, "Unknown")
                    track_identity[track_id] = confirmed_name
                    stats["match"] += time.perf_counter() - match_start

                    should_log = (
                        confirmed_name != previous_name or
                        current_time - track_last_log_time.get(track_id, 0.0) >= LOG_INTERVAL
                    )
                    if should_log:
                        write_event(
                            track_id, confirmed_name, best_similarity, second_similarity,
                            "confirmed" if confirmed_name != "Unknown" else "unknown_or_unconfirmed",
                            quality.reason, person["box"],
                        )

                            f"[{timestamp_now()}] Track {track_id} | "
                            f"Identity: {confirmed_name} | Similarity: {best_similarity:.3f} | "
                            f"Margin OK: {margin_ok}"
                        )
                        track_last_log_time[track_id] = current_time

                stats["face_total"] += time.perf_counter() - face_pass_start

            # Expire identity if no acceptable face has been seen recently.
            for track_id, last_seen in list(track_last_face_time.items()):
                if current_time - last_seen > IDENTITY_TTL_SECONDS:
                    if track_identity.get(track_id) != "Unknown":
                        track_identity[track_id] = "Unknown"
                        match_history[track_id].clear()

            misc_start = time.perf_counter()
            for person in people:
                track_id = person["track_id"]
                x1, y1, x2, y2 = person["box"]
                identity = track_identity.get(track_id, "Unknown")
                label = f"Track {track_id} | {identity}"
                cv2.rectangle(
                    frame, (x1, y1), (x2, y2),
                    (0, 255, 0) if identity != "Unknown" else (0, 165, 255), 2,
                )
                cv2.putText(
                    frame, label, (x1, max(25, y1 - 10)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.65, (255, 255, 255), 2, cv2.LINE_AA,
                )

            elapsed = max(time.perf_counter() - start_time, 0.001)
            processing_fps = frame_count / elapsed
            cv2.putText(
                frame, f"Camera: {CAMERA_ID} | FPS: {processing_fps:.1f}",
                (15, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.65,
                (255, 255, 255), 2, cv2.LINE_AA,
            )
            cv2.imshow("Live Staff Recognition", frame)
            key = cv2.waitKey(1) & 0xFF
            stats["misc"] += time.perf_counter() - misc_start

            now = time.perf_counter()
            if now - last_profile_time >= PROFILE_INTERVAL:
                interval_elapsed = now - last_profile_time
                interval_frames = frame_count - last_profile_frame
                interval_faces = face_pass_count - last_profile_face_pass
                print_profile(stats, interval_elapsed, interval_frames, interval_faces)
                stats.clear()
                last_profile_time = now
                last_profile_frame = frame_count
                last_profile_face_pass = face_pass_count

            if key == ord("q"):
                break

    finally:
        cap.release()
        cv2.destroyAllWindows()
        elapsed = max(time.perf_counter() - start_time, 0.001)
        print("\n" + "=" * 65)
        print("LIVE RECOGNITION STOPPED")
        print(f"Frames processed: {frame_count}")
        print(f"Face recognition passes: {face_pass_count}")
        print(f"Total face-detector calls: {total_face_calls}")
        print(f"Elapsed seconds: {elapsed:.1f}")
        print(f"Average end-to-end FPS: {frame_count / elapsed:.2f}")
        print(f"CSV log: {CSV_PATH}")
        print("=" * 65)

    main()

def timestamp_now():

def clamp_box(box, width, height):
    x1, y1, x2, y2 = map(int, box)
    x1 = max(0, min(x1, width - 1))
    y1 = max(0, min(y1, height - 1))
    x2 = max(0, min(x2, width))
    y2 = max(0, min(y2, height))
    return (x1, y1, x2, y2) if x2 > x1 and y2 > y1 else None

class CameraWorker(threading.Thread):
    def __init__(self, camera_id, rtsp_url, gallery, face_recognizer,
                 quality_checker):
        super().__init__(daemon=True, name=camera_id)
        self.camera_id = camera_id
        self.rtsp_url = rtsp_url
        self.gallery = gallery
        self.face_recognizer = face_recognizer
        self.quality_checker = quality_checker
        self.person_model = None
        self.cap = None
        self.latest_frame = None
        self.frame_lock = threading.Lock()
        self.track_identity = {}
        self.match_history = defaultdict(lambda: deque(maxlen=MATCH_WINDOW))
        self.track_last_seen = {}
        self.track_last_face_attempt = {}
        self.track_last_log = {}
        self.frame_count = 0
        self.error = None

    def connect(self):
        params = [
            cv2.CAP_PROP_OPEN_TIMEOUT_MSEC, 10000,
            cv2.CAP_PROP_READ_TIMEOUT_MSEC, 10000,
        ]
        self.cap = cv2.VideoCapture()
        if not self.cap.open(self.rtsp_url, cv2.CAP_FFMPEG, params):
            self.cap.release()
            self.cap = None
            raise RuntimeError(f"{self.camera_id}: unable to open RTSP stream")

    def recognize_crop(self, crop):
        # One shared InsightFace instance is protected against concurrent calls.
        with FACE_LOCK:
            faces = self.face_recognizer.app.get(crop)
        if len(faces) != 1:
            return None

        face = faces[0]
        quality = self.quality_checker.evaluate(crop, face)
        if not quality.acceptable:
            return None

        embedding = self.face_recognizer.normalize_embedding(face.normed_embedding)
        ranked = self.face_recognizer.rank_candidates(embedding, self.gallery)
        if not ranked:
            return None

        best = ranked[0]
        best_name = best["name"]
        similarity = float(best["similarity"])
        second = float(ranked[1]["similarity"]) if len(ranked) > 1 else None
        margin_ok = second is None or similarity - second >= MIN_SIMILARITY_MARGIN
        if similarity < FACE_SIMILARITY_THRESHOLD or not margin_ok:
            # Ambiguous/weak matches do not vote for Unknown, and do not refresh
            # identity confirmation.
            return None
        return best_name, similarity, second

    def process_face(self, frame, person, now):
        track_id = person["track_id"]
        if now - self.track_last_face_attempt.get(track_id, 0.0) < FACE_PROCESS_INTERVAL:
            return
        self.track_last_face_attempt[track_id] = now

        x1, y1, x2, y2 = person["box"]
        pw, ph = x2 - x1, y2 - y1
        if pw < MIN_PERSON_CROP_WIDTH or ph < MIN_PERSON_CROP_HEIGHT:
            return

        crop_y2 = y1 + int(ph * PERSON_CROP_BOTTOM_RATIO)
        crop = frame[y1:crop_y2, x1:x2]
        if crop.size == 0:
            return

        result = self.recognize_crop(crop)
        # No face, low-quality face, or ambiguous match is not a negative identity vote.
        if result is None:
            return

        name, similarity, second = result
        self.match_history[track_id].append(name)
        votes = list(self.match_history[track_id])
        counts = {candidate: votes.count(candidate) for candidate in set(votes)}
        winner = max(counts, key=counts.get)

        previous = self.track_identity.get(track_id, "Unknown")
        if counts[winner] >= MIN_MATCH_VOTES:
            self.track_identity[track_id] = winner

        identity = self.track_identity.get(track_id, "Unknown")
        changed = identity != previous
        log_due = now - self.track_last_log.get(track_id, 0.0) >= 5.0
        if changed or log_due:
            write_event(
                self.camera_id, track_id, identity, similarity, second,
                "confirmed" if identity != "Unknown" else "candidate_not_confirmed",
                person["box"],
            )
            safe_print(
                f"[{timestamp_now()}] {self.camera_id} | Local Track {track_id} "
                f"| Identity: {identity} | Similarity: {similarity:.3f}"
            )
            self.track_last_log[track_id] = now

    def run(self):
        try:
            safe_print(f"[{self.camera_id}] Loading detector...")
            self.person_model = YOLO(PERSON_MODEL)
            safe_print(f"[{self.camera_id}] Connecting...")
            self.connect()
            safe_print(f"[{self.camera_id}] Connected.")

            while not STOP_EVENT.is_set():
                ok, frame = self.cap.read()
                if not ok or frame is None:
                    safe_print(f"[{self.camera_id}] Frame read failed; reconnecting...")
                    if self.cap is not None:
                        self.cap.release()
                    time.sleep(1.0)
                    if STOP_EVENT.is_set():
                        break
                    try:
                        self.connect()
                    except Exception as exc:
                        safe_print(f"[{self.camera_id}] Reconnect failed: {exc}")
                        time.sleep(2.0)
                    continue

                self.frame_count += 1
                now = time.time()
                height, width = frame.shape[:2]
                results = self.person_model.track(
                    frame, persist=True, tracker="bytetrack.yaml",
                    classes=[0], conf=PERSON_CONFIDENCE,
                    imgsz=IMAGE_SIZE, device="cpu", verbose=False,
                )

                people = []
                if results and results[0].boxes is not None:
                    boxes = results[0].boxes
                    if boxes.id is not None:
                        for box, track_id in zip(
                            boxes.xyxy.cpu().numpy(),
                            boxes.id.int().cpu().tolist(),
                        ):
                            person_box = clamp_box(box, width, height)
                            if person_box is None:
                                continue
                            track_id = int(track_id)
                            people.append({"track_id": track_id, "box": person_box})
                            self.track_last_seen[track_id] = now

                for person in people:
                    self.process_face(frame, person, now)

                # Face loss does not clear identity. Only a local track absent
                # for the configured period loses its cached state.
                for track_id, last_seen in list(self.track_last_seen.items()):
                    if now - last_seen > TRACK_STATE_TTL_SECONDS:
                        self.track_identity.pop(track_id, None)
                        self.match_history.pop(track_id, None)
                        self.track_last_seen.pop(track_id, None)
                        self.track_last_face_attempt.pop(track_id, None)
                        self.track_last_log.pop(track_id, None)

                for person in people:
                    track_id = person["track_id"]
                    x1, y1, x2, y2 = person["box"]
                    identity = self.track_identity.get(track_id, "Unknown")
                    color = (0, 200, 0) if identity != "Unknown" else (0, 165, 255)
                    label = f"{self.camera_id} | Track {track_id} | {identity}"
                    cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                    cv2.putText(
                        frame, label, (x1, max(25, y1 - 10)),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255),
                        2, cv2.LINE_AA,
                    )

                cv2.putText(
                    frame, f"{self.camera_id} | people: {len(people)}",
                    (12, 28), cv2.FONT_HERSHEY_SIMPLEX, 0.7,
                    (255, 255, 255), 2, cv2.LINE_AA,
                )
                with self.frame_lock:
                    self.latest_frame = frame.copy()

        except Exception as exc:
            self.error = str(exc)
            safe_print(f"[{self.camera_id}] ERROR: {exc}")
        finally:
            if self.cap is not None:
                self.cap.release()
            safe_print(f"[{self.camera_id}] Worker stopped.")


def side_by_side(left, right):
    if left is None and right is None:
        return None
    if left is None:
        left = right.copy()
        cv2.putText(left, "CAM-202 waiting", (20, 60),
                    cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 2)
    if right is None:
        right = left.copy()
        cv2.putText(right, "CAM-302 waiting", (20, 90),
                    cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 2)

    target_height = min(left.shape[0], right.shape[0])

    def resize_height(img):
        scale = target_height / img.shape[0]
        return cv2.resize(img, (int(img.shape[1] * scale), target_height))

    return cv2.hconcat([resize_height(left), resize_height(right)])

def now_iso() -> str:
    return datetime.now().astimezone().isoformat(timespec="milliseconds")


def normalize(vec: np.ndarray) -> np.ndarray:
    vec = np.asarray(vec, dtype=np.float32).reshape(-1)
    norm = float(np.linalg.norm(vec))
    return vec / norm if norm > 1e-12 else vec


def cosine(a: Optional[np.ndarray], b: Optional[np.ndarray]) -> float:
    if a is None or b is None:
        return -1.0
    a, b = normalize(a), normalize(b)
    if a.shape != b.shape:
        return -1.0
    return float(np.dot(a, b))


class BodyEmbedder:
    """
    Uses torchreid FeatureExtractor only with an explicit checkpoint supplied by
    V4_REID_CHECKPOINT. ImageNet-only weights are not treated as person-ReID.
    """
    def __init__(self):
        self.extractor = None
        if not BODY_ENABLED:
            print("[BODY] Disabled by V4_BODY_REID=0")
            return
        if not REID_CHECKPOINT:
            print("[BODY] No V4_REID_CHECKPOINT supplied. Body-based cross-camera matching is DISABLED.")
            print("[BODY] This is intentional: ImageNet-only features are not person-ReID embeddings.")
            return
        checkpoint = Path(REID_CHECKPOINT).expanduser()
        if not checkpoint.is_file():
            print(f"[BODY] ReID checkpoint does not exist: {checkpoint}")
            print("[BODY] Body-based cross-camera matching is DISABLED.")
            return
        try:
            import torchreid
            self.extractor = torchreid.utils.FeatureExtractor(
                model_name="osnet_x1_0", model_path=str(checkpoint), device="cpu"
            )
            print(f"[BODY] Loaded explicit person-ReID checkpoint: {checkpoint}")
        except Exception as exc:
            self.extractor = None
            print(f"[BODY] Could not load ReID checkpoint: {type(exc).__name__}: {exc}")
            print("[BODY] Body-based cross-camera matching is DISABLED.")

    def embed(self, bgr_crop: np.ndarray) -> Optional[np.ndarray]:
        if self.extractor is None or bgr_crop is None or bgr_crop.size == 0:
            return None
        h, w = bgr_crop.shape[:2]
        if h < 90 or w < 30:
            return None
        try:
            feature = self.extractor(bgr_crop)
            if hasattr(feature, "detach"):
                feature = feature.detach().cpu().numpy()
            return normalize(np.asarray(feature).reshape(-1))
        except Exception as exc:
            print(f"[BODY] Embedding error: {type(exc).__name__}: {exc}")
            return None


@dataclass
class GlobalPerson:
    global_id: str
    created_at: float
    last_seen: float
    body_embedding: Optional[np.ndarray] = None
    body_samples: int = 0
    staff_name: str = "Unknown"
    face_scores: deque = field(default_factory=lambda: deque(maxlen=7))
    face_name_votes: Counter = field(default_factory=Counter)
    face_vote_times: dict[str, deque] = field(default_factory=dict)
    last_face_at: float = 0.0
    cameras_seen: set = field(default_factory=set)


@dataclass
class LocalTrack:
    camera_id: str
    local_id: int
    global_id: str
    last_seen: float
    last_face_attempt: float = 0.0
    last_body_attempt: float = 0.0
    face_similarity: float = -1.0
    body_similarity: float = -1.0
    staff_name: str = "Unknown"
    last_bbox: tuple = (0, 0, 0, 0)
    match_reason: str = "new_global"
    match_score: float = -1.0


class SharedRegistry:
    def __init__(self):
        self.globals: dict[str, GlobalPerson] = {}
        self.locals: dict[tuple[str, int], LocalTrack] = {}
        self.counter = 0
        self.attendance_written: set[tuple[str, str]] = set()

    def _new_global(self, t: float) -> GlobalPerson:
        self.counter += 1
        gid = f"G{self.counter:05d}"
        person = GlobalPerson(gid, t, t)
        self.globals[gid] = person
        return person

    def _active_local_ids(self, camera_id: str, t: float) -> set[str]:
        return {lt.global_id for (cam, _), lt in self.locals.items()
                if cam == camera_id and t - lt.last_seen < 2.0}

    def assign(self, camera_id: str, local_id: int, body: Optional[np.ndarray],
               t: float) -> tuple[LocalTrack, float]:
        key = (camera_id, local_id)
        with state_lock:
            lt = self.locals.get(key)
            if lt is not None:
                lt.last_seen = t
                gp = self.globals.get(lt.global_id)
                if gp is not None:
                    gp.last_seen = t
                    gp.cameras_seen.add(camera_id)
                return lt, lt.match_score

            blocked = self._active_local_ids(camera_id, t)
            best_gp, best_sim = None, -1.0
            reason = "new_global"

            # Body matching only works when an explicit person-ReID checkpoint loaded.
            if body is not None:
                candidates = []
                for gp in self.globals.values():
                    if t - gp.last_seen > GLOBAL_TTL or gp.global_id in blocked:
                        continue
                    if gp.body_embedding is not None:
                        candidates.append((cosine(body, gp.body_embedding), gp))
                candidates.sort(key=lambda item: item[0], reverse=True)
                if candidates:
                    top_sim, top_gp = candidates[0]
                    second_sim = candidates[1][0] if len(candidates) > 1 else -1.0
                    if top_sim >= BODY_MATCH_THRESHOLD and top_sim - second_sim >= 0.04:
                        best_gp, best_sim, reason = top_gp, top_sim, "body_reid"

            if best_gp is None:
                best_gp = self._new_global(t)
                reason = "new_global"
                best_sim = -1.0

            lt = LocalTrack(camera_id, local_id, best_gp.global_id, t,
                            match_reason=reason, match_score=best_sim)
            self.locals[key] = lt
            best_gp.last_seen = t
            best_gp.cameras_seen.add(camera_id)
            return lt, best_sim

    def link_confirmed_face(self, lt: LocalTrack, name: str, t: float) -> LocalTrack:
        """
        Confirmed face names are canonical staff identities. If the same staff
        member has accumulated duplicate global IDs, link to the oldest eligible
        same-name identity, but never merge two tracks active in the same camera.
        """
        if not CROSS_CAMERA_FACE_LINK or name == "Unknown":
            return lt
        with state_lock:
            current = self.globals.get(lt.global_id)
            if current is None or current.staff_name not in ("Unknown", name):
                return lt

            blocked = self._active_local_ids(lt.camera_id, t)
            candidates = [
                gp for gp in self.globals.values()
                if gp.global_id != current.global_id
                and gp.staff_name == name
                and gp.global_id not in blocked
                and t - gp.last_seen <= GLOBAL_TTL
            ]
            if not candidates:
                return lt

            # All candidates already have the same confirmed staff name. Prefer
            # the oldest identity to prevent creating a new canonical ID per visit.
            target = min(candidates, key=lambda gp: gp.created_at)
            old_id = lt.global_id
            lt.global_id = target.global_id
            lt.match_reason = "confirmed_face_canonical_staff"
            lt.match_score = -1.0
            target.last_seen = t
            target.cameras_seen.add(lt.camera_id)

            # Repoint tracks from duplicate IDs on other cameras to the canonical
            # identity. Do not repoint an active track in this camera, because two
            # simultaneous people must never share one global ID.
            duplicate_ids = {
                gp.global_id for gp in candidates
                if gp.global_id != target.global_id
            }
            duplicate_ids.add(old_id)
            for other in self.locals.values():
                if other is lt or other.global_id not in duplicate_ids:
                    continue
                if other.camera_id == lt.camera_id and t - other.last_seen < 2.0:
                    continue
                other.global_id = target.global_id
                other.match_reason = "canonical_staff_merge"
                other.match_score = -1.0

            # Merge metadata and remove duplicate registry entries only when no
            # local tracks still refer to them.
            for duplicate_id in list(duplicate_ids):
                if duplicate_id == target.global_id:
                    continue
                duplicate = self.globals.get(duplicate_id)
                if duplicate is not None:
                    target.cameras_seen.update(duplicate.cameras_seen)
                    if target.body_embedding is None and duplicate.body_embedding is not None:
                        target.body_embedding = duplicate.body_embedding.copy()
                        target.body_samples = duplicate.body_samples
                referenced = any(x.global_id == duplicate_id for x in self.locals.values())
                if not referenced:
                    self.globals.pop(duplicate_id, None)
            return lt

    def safe_get_local(self, camera_id: str, local_id: int) -> Optional[LocalTrack]:
        with state_lock:
            return self.locals.get((camera_id, local_id))

    def update_body(self, lt: LocalTrack, emb: Optional[np.ndarray], t: float):
        if emb is None:
            return
        with state_lock:
            gp = self.globals.get(lt.global_id)
            if gp is None:
                return
            if gp.body_embedding is None:
                gp.body_embedding = emb.copy()
                gp.body_samples = 1
            else:
                # Running mean reduces noise; normalize after combining.
                gp.body_embedding = normalize(gp.body_embedding * min(gp.body_samples, 8) + emb)
                gp.body_samples += 1
            gp.last_seen = t

    def update_face(self, lt: LocalTrack, name: str, similarity: float, t: float):
        lt.face_similarity = similarity
        if name == "Unknown" or similarity < FACE_MATCH_THRESHOLD:
            return
        confirmed_name = "Unknown"
        with state_lock:
            gp = self.globals.get(lt.global_id)
            if gp is None:
                return
            if gp.staff_name not in ("Unknown", name):
                print(f"[SAFETY] Conflicting face names for {gp.global_id}: "
                      f"{gp.staff_name} vs {name}; leaving name unchanged.")
                return
            votes = gp.face_vote_times.setdefault(name, deque())
            votes.append(t)
            while votes and t - votes[0] > FACE_VOTE_WINDOW_SEC:
                votes.popleft()
            gp.face_name_votes[name] = len(votes)
            gp.face_scores.append(similarity)
            gp.last_face_at = t
            if len(votes) >= FACE_VOTES_REQUIRED and len(gp.face_scores) >= 2:
                gp.staff_name = name
                confirmed_name = name
            lt.staff_name = gp.staff_name
        if confirmed_name != "Unknown":
            self.link_confirmed_face(lt, confirmed_name, t)

    def get_name(self, lt: LocalTrack) -> str:
        with state_lock:
            gp = self.globals.get(lt.global_id)
            return gp.staff_name if gp else "Unknown"

    def prune(self, t: float):
        with state_lock:
            expired = [gid for gid, gp in self.globals.items()
                       if t - gp.last_seen > GLOBAL_TTL]
            for gid in expired:
                self.globals.pop(gid, None)
            # Keep stale local track mappings only while their global record exists.
            for key, lt in list(self.locals.items()):
                if lt.global_id not in self.globals and t - lt.last_seen > GLOBAL_TTL:
                    self.locals.pop(key, None)

    def maybe_attendance(self, lt: LocalTrack, camera_id: str):
        name = self.get_name(lt)
        if name == "Unknown":
            return
        day = datetime.now().astimezone().date().isoformat()
        key = (day, name)
        with state_lock:
            if key in self.attendance_written:
                return
            self.attendance_written.add(key)
        event_queue.put(("attendance", {
            "timestamp": now_iso(), "date": day, "staff_name": name,
            "global_id": lt.global_id, "camera_id": camera_id,
            "event_type": "first_seen", "evidence": "repeated_face_match"
        }))

def init_csvs():
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    TRACKER_CONFIG.write_text(
        "tracker_type: bytetrack\\n"
        "track_high_thresh: 0.25\\n"
        "track_low_thresh: 0.10\\n"
        "new_track_thresh: 0.25\\n"
        "track_buffer: 90\\n"
        "match_thresh: 0.80\\n"
        "fuse_score: True\\n",
        encoding="utf-8",
    )
    with TRACK_CSV.open("w", newline="", encoding="utf-8") as f:
        csv.DictWriter(f, fieldnames=[
            "timestamp", "camera_id", "local_track_id", "global_id", "staff_name",
            "face_similarity", "body_similarity", "bbox_x1", "bbox_y1", "bbox_x2",
            "bbox_y2", "status", "match_reason", "match_score", "cameras_seen"
        ]).writeheader()
    with ATTENDANCE_CSV.open("w", newline="", encoding="utf-8") as f:
        csv.DictWriter(f, fieldnames=[
            "timestamp", "date", "staff_name", "global_id", "camera_id",
            "event_type", "evidence"
        ]).writeheader()


def csv_writer():
    while not stop_event.is_set() or not event_queue.empty():
        try:
            kind, row = event_queue.get(timeout=0.25)
        except queue.Empty:
            continue
        path = TRACK_CSV if kind == "track" else ATTENDANCE_CSV
        with csv_lock, path.open("a", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=row.keys())
            writer.writerow(row)
        event_queue.task_done()


def crop_box(frame: np.ndarray, xyxy) -> Optional[np.ndarray]:
    h, w = frame.shape[:2]
    x1, y1, x2, y2 = [int(v) for v in xyxy]
    x1, y1 = max(0, x1), max(0, y1)
    x2, y2 = min(w, x2), min(h, y2)
    if x2 <= x1 or y2 <= y1:
        return None
    return frame[y1:y2, x1:x2].copy()


def recognize_face(face_recognizer, crop: Optional[np.ndarray], gallery):
    if crop is None or crop.size == 0 or min(crop.shape[:2]) < 80:
        return "Unknown", -1.0
    try:
        # InsightFace app is shared between camera threads, so serialize calls.
        with face_lock:
            faces = face_recognizer.app.get(crop)
        if not faces:
            return "Unknown", -1.0
        # Use the largest face in the person crop.
        face = max(faces, key=lambda f: max(0, f.bbox[2] - f.bbox[0]) *
                   max(0, f.bbox[3] - f.bbox[1]))
        embedding = face_recognizer.normalize_embedding(face.normed_embedding)
        candidates = face_recognizer.rank_candidates(embedding, gallery)
        if not candidates:
            return "Unknown", -1.0
        best = candidates[0]
        second = candidates[1]["similarity"] if len(candidates) > 1 else -1.0
        score = float(best["similarity"])
        if score >= FACE_MATCH_THRESHOLD and (score - float(second)) >= FACE_MARGIN:
            return str(best["name"]), score
        return "Unknown", score
    except Exception as exc:
        print(f"[FACE] Recognition error: {exc}")
        return "Unknown", -1.0


def camera_worker(camera_id: str, url: str, face_recognizer, gallery, embedder):
    print(f"[{camera_id}] Starting camera worker")
    model = YOLO(YOLO_MODEL)
    tracks_last_seen = {}
    cap = None
    frame_no = 0
    try:
        while not stop_event.is_set():
            if cap is None or not cap.isOpened():
                if cap is not None:
                    cap.release()
                cap = cv2.VideoCapture(url, cv2.CAP_FFMPEG)
                cap.set(cv2.CAP_PROP_BUFFERSIZE, 2)
                if not cap.isOpened():
                    print(f"[{camera_id}] RTSP open failed; retrying in 3s")
                    time.sleep(3)
                    continue
            ok, frame = cap.read()
            if not ok or frame is None:
                print(f"[{camera_id}] Read failed; reconnecting")
                cap.release()
                cap = None
                time.sleep(1)
                continue
            frame_no += 1
            t = time.monotonic()
            try:
                result = model.track(frame, persist=True, tracker=str(TRACKER_CONFIG),
                                     classes=[0], conf=YOLO_CONF, imgsz=YOLO_IMGSZ,
                                     verbose=False, device="cpu")[0]
            except Exception as exc:
                print(f"[{camera_id}] YOLO error: {exc}")
                continue
            boxes = result.boxes
            if boxes is None or boxes.id is None:
                registry.prune(t)
                continue

            ids = boxes.id.int().cpu().tolist()
            coords = boxes.xyxy.cpu().numpy()
            for local_id, xyxy in zip(ids, coords):
                crop = crop_box(frame, xyxy)
                if crop is None:
                    continue
                key = (camera_id, int(local_id))
                lt = registry.safe_get_local(camera_id, int(local_id))
                body_emb = None
                if lt is None or t - lt.last_body_attempt >= 1.5:
                    body_emb = embedder.embed(crop)
                    # Existing local track mapping can be read by camera thread; registry writes lock.
                    if lt is not None:
                        lt.last_body_attempt = t
                lt, match_sim = registry.assign(camera_id, int(local_id), body_emb, t)
                lt.body_similarity = match_sim if match_sim >= 0 else lt.body_similarity
                if body_emb is not None:
                    registry.update_body(lt, body_emb, t)
                    lt.last_body_attempt = t

                if t - lt.last_face_attempt >= FACE_INTERVAL:
                    name, score = recognize_face(face_recognizer, crop, gallery)
                    registry.update_face(lt, name, score, t)
                    lt.last_face_attempt = t

                staff_name = registry.get_name(lt)
                lt.staff_name = staff_name
                registry.maybe_attendance(lt, camera_id)
                x1, y1, x2, y2 = [int(v) for v in xyxy]
                color = (0, 200, 0) if staff_name != "Unknown" else (0, 180, 255)
                label = f"{staff_name} | {lt.global_id} | L{local_id}"
                cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                cv2.putText(frame, label, (x1, max(22, y1 - 8)),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.55, color, 2)
                event_queue.put(("track", {
                    "timestamp": now_iso(), "camera_id": camera_id,
                    "local_track_id": int(local_id), "global_id": lt.global_id,
                    "staff_name": staff_name,
                    "face_similarity": round(lt.face_similarity, 4),
                    "body_similarity": round(lt.body_similarity, 4),
                    "bbox_x1": x1, "bbox_y1": y1,  "bbox_x2": x2, "bbox_y2": y2,
                    "status": "identified" if staff_name != "Unknown" else "tracking",
                    "match_reason": lt.match_reason,
                    "match_score": round(lt.match_score, 4),
                    "cameras_seen": ",".join(sorted(registry.globals.get(lt.global_id).cameras_seen))
                        if registry.globals.get(lt.global_id) else camera_id
                }))
            registry.prune(t)
            cv2.imshow(f"V4 {camera_id}", cv2.resize(frame, (960, 540)))
            if cv2.waitKey(1) & 0xFF == ord("q"):
                stop_event.set()
                break
    except Exception as exc:
        print(f"[{camera_id}] Worker stopped after error: {type(exc).__name__}: {exc}")
    finally:
        if cap is not None:
            cap.release()
        try:
            cv2.destroyWindow(f"V4 {camera_id}")
        except Exception:
            pass
        print(f"[{camera_id}] Stopped")


def main():
    configured = {k: v for k, v in CAMERAS.items() if v.strip()}
    if not configured:
        print("No camera URLs found. Set CAMERA_202_RTSP and/or CAMERA_302_RTSP in PowerShell.")
        return 2
    if not Path(YOLO_MODEL).exists() and not Path(YOLO_MODEL).is_absolute():
        print(f"YOLO model not found: {YOLO_MODEL}")
        return 2
    if not GALLERY_PATH.exists():
        print(f"Staff gallery not found: {GALLERY_PATH}")
        return 2

    init_csvs()
    writer_thread = threading.Thread(target=csv_writer, daemon=True)
    writer_thread.start()

    face_recognizer = FaceRecognizer()
    gallery = face_recognizer.load_gallery(str(GALLERY_PATH))
    if not gallery:
        print(f"No staff faces loaded from {GALLERY_PATH}")
        stop_event.set()
        writer_thread.join(timeout=2)
        return 2
    print(f"[FACE] Gallery identities: {', '.join(gallery.keys())}")

    embedder = BodyEmbedder()
    threads = []
    for camera_id, url in configured.items():
        th = threading.Thread(target=camera_worker,
                              args=(camera_id, url, face_recognizer, gallery, embedder),
                              daemon=True, name=camera_id)
        th.start()
        threads.append(th)
    print("V4 running. Press q in a video window or Ctrl+C to stop.")
    print(f"[CONFIG] body_reid={'enabled' if embedder.extractor is not None else 'disabled'}; "
          f"face_votes_required={FACE_VOTES_REQUIRED}; face_link={CROSS_CAMERA_FACE_LINK}")
    try:
        while any(th.is_alive() for th in threads):
            time.sleep(0.5)
    except KeyboardInterrupt:
        print("Stopping...")
        stop_event.set()
    finally:
        stop_event.set()
        for th in threads:
            th.join(timeout=5)
        event_queue.join()
        writer_thread.join(timeout=2)
        cv2.destroyAllWindows()
    print(f"Track log: {TRACK_CSV}")
    print(f"First-seen attendance log: {ATTENDANCE_CSV}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def now_iso() -> str:
    return datetime.now().astimezone().isoformat(timespec="milliseconds")


def normalize(vec: np.ndarray) -> np.ndarray:
    vec = np.asarray(vec, dtype=np.float32).reshape(-1)
    norm = float(np.linalg.norm(vec))
    return vec / norm if norm > 1e-12 else vec


def cosine(a: Optional[np.ndarray], b: Optional[np.ndarray]) -> float:
    if a is None or b is None:
        return -1.0
    a, b = normalize(a), normalize(b)
    if a.shape != b.shape:
        return -1.0
    return float(np.dot(a, b))


class BodyEmbedder:
    """
    Uses torchreid FeatureExtractor only with an explicit checkpoint supplied by
    V4_REID_CHECKPOINT. ImageNet-only weights are not treated as person-ReID.
    """
    def __init__(self):
        self.extractor = None
        if not BODY_ENABLED:
            print("[BODY] Disabled by V4_BODY_REID=0")
            return
        if not REID_CHECKPOINT:
            print("[BODY] No V4_REID_CHECKPOINT supplied. Body-based cross-camera matching is DISABLED.")
            print("[BODY] This is intentional: ImageNet-only features are not person-ReID embeddings.")
            return
        checkpoint = Path(REID_CHECKPOINT).expanduser()
        if not checkpoint.is_file():
            print(f"[BODY] ReID checkpoint does not exist: {checkpoint}")
            print("[BODY] Body-based cross-camera matching is DISABLED.")
            return
        try:
            import torchreid
            self.extractor = torchreid.utils.FeatureExtractor(
                model_name="osnet_x1_0", model_path=str(checkpoint), device="cpu"
            )
            print(f"[BODY] Loaded explicit person-ReID checkpoint: {checkpoint}")
        except Exception as exc:
            self.extractor = None
            print(f"[BODY] Could not load ReID checkpoint: {type(exc).__name__}: {exc}")
            print("[BODY] Body-based cross-camera matching is DISABLED.")

    def embed(self, bgr_crop: np.ndarray) -> Optional[np.ndarray]:
        if self.extractor is None or bgr_crop is None or bgr_crop.size == 0:
            return None
        h, w = bgr_crop.shape[:2]
        if h < 90 or w < 30:
            return None
        try:
            feature = self.extractor(bgr_crop)
            if hasattr(feature, "detach"):
                feature = feature.detach().cpu().numpy()
            return normalize(np.asarray(feature).reshape(-1))
        except Exception as exc:
            print(f"[BODY] Embedding error: {type(exc).__name__}: {exc}")
            return None


@dataclass
class GlobalPerson:
    global_id: str
    created_at: float
    last_seen: float
    body_embedding: Optional[np.ndarray] = None
    body_samples: int = 0
    staff_name: str = "Unknown"
    face_scores: deque = field(default_factory=lambda: deque(maxlen=7))
    face_name_votes: Counter = field(default_factory=Counter)
    face_vote_times: dict[str, deque] = field(default_factory=dict)
    last_face_at: float = 0.0
    cameras_seen: set = field(default_factory=set)


@dataclass
class LocalTrack:
    camera_id: str
    local_id: int
    global_id: str
    last_seen: float
    last_face_attempt: float = 0.0
    last_body_attempt: float = 0.0
    face_similarity: float = -1.0
    body_similarity: float = -1.0
    staff_name: str = "Unknown"
    last_bbox: tuple = (0, 0, 0, 0)
    match_reason: str = "new_global"
    match_score: float = -1.0


class SharedRegistry:
    def __init__(self):
        self.globals: dict[str, GlobalPerson] = {}
        self.locals: dict[tuple[str, int], LocalTrack] = {}
        self.counter = 0
        self.attendance_written: set[tuple[str, str]] = set()

    def _new_global(self, t: float) -> GlobalPerson:
        self.counter += 1
        gid = f"G{self.counter:05d}"
        person = GlobalPerson(gid, t, t)
        self.globals[gid] = person
        return person

    def _active_local_ids(self, camera_id: str, t: float) -> set[str]:
        return {lt.global_id for (cam, _), lt in self.locals.items()
                if cam == camera_id and t - lt.last_seen < 2.0}

    def assign(self, camera_id: str, local_id: int, body: Optional[np.ndarray],
               t: float) -> tuple[LocalTrack, float]:
        key = (camera_id, local_id)
        with state_lock:
            lt = self.locals.get(key)
            if lt is not None:
                lt.last_seen = t
                gp = self.globals.get(lt.global_id)
                if gp is not None:
                    gp.last_seen = t
                    gp.cameras_seen.add(camera_id)
                return lt, lt.match_score

            blocked = self._active_local_ids(camera_id, t)
            best_gp, best_sim = None, -1.0
            reason = "new_global"

            # Body matching only works when an explicit person-ReID checkpoint loaded.
            if body is not None:
                candidates = []
                for gp in self.globals.values():
                    if t - gp.last_seen > GLOBAL_TTL or gp.global_id in blocked:
                        continue
                    if gp.body_embedding is not None:
                        candidates.append((cosine(body, gp.body_embedding), gp))
                candidates.sort(key=lambda item: item[0], reverse=True)
                if candidates:
                    top_sim, top_gp = candidates[0]
                    second_sim = candidates[1][0] if len(candidates) > 1 else -1.0
                    if top_sim >= BODY_MATCH_THRESHOLD and top_sim - second_sim >= 0.04:
                        best_gp, best_sim, reason = top_gp, top_sim, "body_reid"

            if best_gp is None:
                best_gp = self._new_global(t)
                reason = "new_global"
                best_sim = -1.0

            lt = LocalTrack(camera_id, local_id, best_gp.global_id, t,
                            match_reason=reason, match_score=best_sim)
            self.locals[key] = lt
            best_gp.last_seen = t
            best_gp.cameras_seen.add(camera_id)
            return lt, best_sim

    def link_confirmed_face(self, lt: LocalTrack, name: str, t: float) -> LocalTrack:
        """
        Confirmed face names are canonical staff identities. If the same staff
        member has accumulated duplicate global IDs, link to the oldest eligible
        same-name identity, but never merge two tracks active in the same camera.
        """
        if not CROSS_CAMERA_FACE_LINK or name == "Unknown":
            return lt
        with state_lock:
            current = self.globals.get(lt.global_id)
            if current is None or current.staff_name not in ("Unknown", name):
                return lt

            blocked = self._active_local_ids(lt.camera_id, t)
            candidates = [
                gp for gp in self.globals.values()
                if gp.global_id != current.global_id
                and gp.staff_name == name
                and gp.global_id not in blocked
                and t - gp.last_seen <= GLOBAL_TTL
            ]
            if not candidates:
                return lt

            # All candidates already have the same confirmed staff name. Prefer
            # the oldest identity to prevent creating a new canonical ID per visit.
            target = min(candidates, key=lambda gp: gp.created_at)
            old_id = lt.global_id
            lt.global_id = target.global_id
            lt.match_reason = "confirmed_face_canonical_staff"
            lt.match_score = -1.0
            target.last_seen = t
            target.cameras_seen.add(lt.camera_id)

            # Repoint tracks from duplicate IDs on other cameras to the canonical
            # identity. Do not repoint an active track in this camera, because two
            # simultaneous people must never share one global ID.
            duplicate_ids = {
                gp.global_id for gp in candidates
                if gp.global_id != target.global_id
            }
            duplicate_ids.add(old_id)
            for other in self.locals.values():
                if other is lt or other.global_id not in duplicate_ids:
                    continue
                if other.camera_id == lt.camera_id and t - other.last_seen < 2.0:
                    continue
                other.global_id = target.global_id
                other.match_reason = "canonical_staff_merge"
                other.match_score = -1.0

            # Merge metadata and remove duplicate registry entries only when no
            # local tracks still refer to them.
            for duplicate_id in list(duplicate_ids):
                if duplicate_id == target.global_id:
                    continue
                duplicate = self.globals.get(duplicate_id)
                if duplicate is not None:
                    target.cameras_seen.update(duplicate.cameras_seen)
                    if target.body_embedding is None and duplicate.body_embedding is not None:
                        target.body_embedding = duplicate.body_embedding.copy()
                        target.body_samples = duplicate.body_samples
                referenced = any(x.global_id == duplicate_id for x in self.locals.values())
                if not referenced:
                    self.globals.pop(duplicate_id, None)
            return lt

    def safe_get_local(self, camera_id: str, local_id: int) -> Optional[LocalTrack]:
        with state_lock:
            return self.locals.get((camera_id, local_id))

    def update_body(self, lt: LocalTrack, emb: Optional[np.ndarray], t: float):
        if emb is None:
            return
        with state_lock:
            gp = self.globals.get(lt.global_id)
            if gp is None:
                return
            if gp.body_embedding is None:
                gp.body_embedding = emb.copy()
                gp.body_samples = 1
            else:
                # Running mean reduces noise; normalize after combining.
                gp.body_embedding = normalize(gp.body_embedding * min(gp.body_samples, 8) + emb)
                gp.body_samples += 1
            gp.last_seen = t

    def update_face(self, lt: LocalTrack, name: str, similarity: float, t: float):
        lt.face_similarity = similarity
        if name == "Unknown" or similarity < FACE_MATCH_THRESHOLD:
            return
        confirmed_name = "Unknown"
        with state_lock:
            gp = self.globals.get(lt.global_id)
            if gp is None:
                return
            if gp.staff_name not in ("Unknown", name):
                print(f"[SAFETY] Conflicting face names for {gp.global_id}: "
                      f"{gp.staff_name} vs {name}; leaving name unchanged.")
                return
            votes = gp.face_vote_times.setdefault(name, deque())
            votes.append(t)
            while votes and t - votes[0] > FACE_VOTE_WINDOW_SEC:
                votes.popleft()
            gp.face_name_votes[name] = len(votes)
            gp.face_scores.append(similarity)
            gp.last_face_at = t
            if len(votes) >= FACE_VOTES_REQUIRED and len(gp.face_scores) >= 2:
                gp.staff_name = name
                confirmed_name = name
            lt.staff_name = gp.staff_name
        if confirmed_name != "Unknown":
            self.link_confirmed_face(lt, confirmed_name, t)

    def get_name(self, lt: LocalTrack) -> str:
        with state_lock:
            gp = self.globals.get(lt.global_id)
            return gp.staff_name if gp else "Unknown"

    def prune(self, t: float):
        with state_lock:
            expired = [gid for gid, gp in self.globals.items()
                       if t - gp.last_seen > GLOBAL_TTL]
            for gid in expired:
                self.globals.pop(gid, None)
            # Keep stale local track mappings only while their global record exists.
            for key, lt in list(self.locals.items()):
                if lt.global_id not in self.globals and t - lt.last_seen > GLOBAL_TTL:
                    self.locals.pop(key, None)

    def maybe_attendance(self, lt: LocalTrack, camera_id: str):
        name = self.get_name(lt)
        if name == "Unknown":
            return
        day = datetime.now().astimezone().date().isoformat()
        key = (day, name)
        with state_lock:
            if key in self.attendance_written:
                return
            self.attendance_written.add(key)
        event_queue.put(("attendance", {
            "timestamp": now_iso(), "date": day, "staff_name": name,
            "global_id": lt.global_id, "camera_id": camera_id,
            "event_type": "first_seen", "evidence": "repeated_face_match"
        }))

def csv_writer():
    while not stop_event.is_set() or not event_queue.empty():
        try:
            kind, row = event_queue.get(timeout=0.25)
        except queue.Empty:
            continue
        path = TRACK_CSV if kind == "track" else ATTENDANCE_CSV
        with csv_lock, path.open("a", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=row.keys())
            writer.writerow(row)
        event_queue.task_done()


def crop_box(frame: np.ndarray, xyxy) -> Optional[np.ndarray]:
    h, w = frame.shape[:2]
    x1, y1, x2, y2 = [int(v) for v in xyxy]
    x1, y1 = max(0, x1), max(0, y1)
    x2, y2 = min(w, x2), min(h, y2)
    if x2 <= x1 or y2 <= y1:
        return None
    return frame[y1:y2, x1:x2].copy()


def recognize_face(face_recognizer, crop: Optional[np.ndarray], gallery):
    if crop is None or crop.size == 0 or min(crop.shape[:2]) < 80:
        return "Unknown", -1.0
    try:
        # InsightFace app is shared between camera threads, so serialize calls.
        with face_lock:
            faces = face_recognizer.app.get(crop)
        if not faces:
            return "Unknown", -1.0
        # Use the largest face in the person crop.
        face = max(faces, key=lambda f: max(0, f.bbox[2] - f.bbox[0]) *
                   max(0, f.bbox[3] - f.bbox[1]))
        embedding = face_recognizer.normalize_embedding(face.normed_embedding)
        candidates = face_recognizer.rank_candidates(embedding, gallery)
        if not candidates:
            return "Unknown", -1.0
        best = candidates[0]
        second = candidates[1]["similarity"] if len(candidates) > 1 else -1.0
        score = float(best["similarity"])
        if score >= FACE_MATCH_THRESHOLD and (score - float(second)) >= FACE_MARGIN:
            return str(best["name"]), score
        return "Unknown", score
    except Exception as exc:
        print(f"[FACE] Recognition error: {exc}")
        return "Unknown", -1.0


def stream_candidates(source: str) -> list[str]:
    """Build likely stream endpoints from a direct URL or an IP Webcam base URL."""
    source = source.strip().rstrip("/")
    if not source:
        return []
    # A URL already containing a path is tried first as supplied.
    candidates = [source]
    try:
        from urllib.parse import urlparse
        parsed = urlparse(source)
        if parsed.scheme in ("http", "https") and parsed.hostname:
            # Common Android IP Webcam / generic MJPEG endpoints.
            for path in ("/video", "/videofeed", "/mjpegfeed", "/video.mjpg", "/stream"):
                candidate = f"{parsed.scheme}://{parsed.netloc}{path}"
                if candidate not in candidates:
                    candidates.append(candidate)
            # IP Webcam's RTSP endpoint commonly uses the same host/port.
            if parsed.port:
                rtsp = f"rtsp://{parsed.hostname}:{parsed.port}/h264.sdp"
                if rtsp not in candidates:
                    candidates.append(rtsp)
    except Exception:
        pass
    return candidates


def open_stream_auto(camera_id: str, source: str):
    """Try direct URL and common endpoints; return (capture, working_url) or (None, None)."""
    for candidate in stream_candidates(source):
        cap = None
        try:
            cap = cv2.VideoCapture(candidate, cv2.CAP_FFMPEG)
            cap.set(cv2.CAP_PROP_BUFFERSIZE, 2)
            # A successful open alone is insufficient; verify that frames arrive.
            if cap.isOpened():
                ok, frame = cap.read()
                if ok and frame is not None and frame.size:
                    print(f"[{camera_id}] Connected using discovered stream endpoint: {candidate}")
                    return cap, candidate, frame
        except Exception as exc:
            print(f"[{camera_id}] Endpoint probe failed ({candidate}): {type(exc).__name__}: {exc}")
        if cap is not None:
            cap.release()
    print(f"[{camera_id}] Could not open a stream from base URL: {source}")
    print(f"[{camera_id}] Checked endpoints: {', '.join(stream_candidates(source))}")
    return None, None, None


def camera_worker(camera_id: str, url: str, face_recognizer, gallery, embedder):
    print(f"[{camera_id}] Starting camera worker")
    model = YOLO(YOLO_MODEL)
    tracks_last_seen = {}
    cap = None
    frame_no = 0
    try:
        while not stop_event.is_set():
            if cap is None or not cap.isOpened():
                if cap is not None:
                    cap.release()
                cap, active_url, first_frame = open_stream_auto(camera_id, url)
                if cap is None:
                    print(f"[{camera_id}] Stream discovery failed; retrying in 5s")
                    time.sleep(5)
                    continue
                # Process the frame already read during endpoint probing.
                ok, frame = True, first_frame
            else:
                ok, frame = cap.read()
            if not ok or frame is None:
                print(f"[{camera_id}] Read failed; reconnecting")
                cap.release()
                cap = None
                time.sleep(1)
                continue
            frame_no += 1
            t = time.monotonic()
            try:
                result = model.track(frame, persist=True, tracker=str(TRACKER_CONFIG),
                                     classes=[0], conf=YOLO_CONF, imgsz=YOLO_IMGSZ,
                                     verbose=False, device="cpu")[0]
            except Exception as exc:
                print(f"[{camera_id}] YOLO error: {exc}")
                continue
            boxes = result.boxes
            if boxes is None or boxes.id is None:
                registry.prune(t)
                continue

            ids = boxes.id.int().cpu().tolist()
            coords = boxes.xyxy.cpu().numpy()
            for local_id, xyxy in zip(ids, coords):
                crop = crop_box(frame, xyxy)
                if crop is None:
                    continue
                key = (camera_id, int(local_id))
                lt = registry.safe_get_local(camera_id, int(local_id))
                body_emb = None
                if lt is None or t - lt.last_body_attempt >= 1.5:
                    body_emb = embedder.embed(crop)
                    # Existing local track mapping can be read by camera thread; registry writes lock.
                    if lt is not None:
                        lt.last_body_attempt = t
                lt, match_sim = registry.assign(camera_id, int(local_id), body_emb, t)
                lt.body_similarity = match_sim if match_sim >= 0 else lt.body_similarity
                if body_emb is not None:
                    registry.update_body(lt, body_emb, t)
                    lt.last_body_attempt = t

                if t - lt.last_face_attempt >= FACE_INTERVAL:
                    name, score = recognize_face(face_recognizer, crop, gallery)
                    registry.update_face(lt, name, score, t)
                    lt.last_face_attempt = t

                staff_name = registry.get_name(lt)
                lt.staff_name = staff_name
                registry.maybe_attendance(lt, camera_id)
                x1, y1, x2, y2 = [int(v) for v in xyxy]
                color = (0, 200, 0) if staff_name != "Unknown" else (0, 180, 255)
                label = f"{staff_name} | {lt.global_id} | L{local_id}"
                cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                cv2.putText(frame, label, (x1, max(22, y1 - 8)),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.55, color, 2)
                event_queue.put(("track", {
                    "timestamp": now_iso(), "camera_id": camera_id,
                    "local_track_id": int(local_id), "global_id": lt.global_id,
                    "staff_name": staff_name,
                    "face_similarity": round(lt.face_similarity, 4),
                    "body_similarity": round(lt.body_similarity, 4),
                    "bbox_x1": x1, "bbox_y1": y1,  "bbox_x2": x2, "bbox_y2": y2,
                    "status": "identified" if staff_name != "Unknown" else "tracking",
                    "match_reason": lt.match_reason,

from pathlib import Path

import cv2
import numpy as np


def align_face(
    image: np.ndarray,
    landmarks: np.ndarray,
    output_size: tuple[int, int] = (112, 112),
) -> np.ndarray | None:
    """
    Align a face using the 68-point facial landmarks.

    Args:
        image: Original BGR image.
        landmarks: 68 facial landmark points, shape (68, 2).
        output_size: (width, height) of the aligned face.

    Returns:
        Aligned BGR face image, or None if alignment fails.
    """
    points = np.asarray(landmarks, dtype=np.float32).reshape(-1, 2)

    if len(points) != 68:
        raise ValueError(f"Expected 68 landmarks, got {len(points)}")

    if not np.isfinite(points).all():
        return None

    # Standard eye-center locations for a 112x112 aligned face.
    # Landmark indices follow the conventional 68-point layout.
    left_eye = points[36:42].mean(axis=0)
    right_eye = points[42:48].mean(axis=0)

    source = np.array([left_eye, right_eye], dtype=np.float32)

    width, height = output_size

    destination = np.array(
        [
            [0.30 * width, 0.38 * height],
            [0.70 * width, 0.38 * height],
        ],
        dtype=np.float32,
    )

    transform, _ = cv2.estimateAffinePartial2D(
        source,
        destination,
        method=cv2.LMEDS,
    )

    if transform is None:
        return None

    aligned = cv2.warpAffine(
        image,
        transform,
        (width, height),
        flags=cv2.INTER_LINEAR,
        borderMode=cv2.BORDER_CONSTANT,
    )

    return aligned

class FaceDetector:
    """Detect faces using OpenCV Haar cascades with an optional YOLO face model."""

    def __init__(self, model_path=None, scale_factor=1.1, min_neighbors=5, confidence=0.5):
        self.model_path = model_path
        self.scale_factor = scale_factor
        self.min_neighbors = min_neighbors
        self.confidence = confidence
        self._model = None

        if model_path:
            try:
                from ultralytics import YOLO

                self._model = YOLO(model_path)
            except Exception:
                self._model = None

        if self._model is None and hasattr(cv2, "CascadeClassifier"):
            cascade_path = cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
            self._model = cv2.CascadeClassifier(cascade_path)

    def detect(self, image):
        if image is None or image.size == 0:
            raise ValueError("Input image is empty or invalid.")

        image = np.asarray(image)
        if image.ndim == 2:
            image = cv2.cvtColor(image, cv2.COLOR_GRAY2BGR)

        if self.model_path and self._model is not None and hasattr(self._model, "predict"):
            results = self._model.predict(
                source=image,
                conf=self.confidence,
                imgsz=640,
                device="cpu",
                verbose=False,
            )

            detections = []
            for result in results:
                if result.boxes is None:
                    continue
                for box in result.boxes:
                    x1, y1, x2, y2 = box.xyxy[0].cpu().numpy().tolist()
                    detections.append({
                        "bbox": (int(x1), int(y1), int(x2), int(y2)),
                        "confidence": float(box.conf[0].cpu().item()),
                        "class_id": int(box.cls[0].cpu().item()) if box.cls is not None else 0,
                    })
            return detections

        if self._model is None:
            return []

        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        if not hasattr(self._model, "detectMultiScale"):
            return []

        detections = self._model.detectMultiScale(
            gray,
            scaleFactor=self.scale_factor,
            minNeighbors=self.min_neighbors,
        )

        results = []
        for x, y, width, height in detections:
            x1, y1, x2, y2 = x, y, x + width, y + height
            results.append({
                "bbox": (int(x1), int(y1), int(x2), int(y2)),
                "confidence": 1.0,
                "class_id": 0,
            })
        return results

    def draw_detections(self, image, detections):
        output = image.copy()
        for detection in detections:

        return output

class FacePersonAssociator:
    """
    Associate detected faces with person bounding boxes.

    Initial implementation uses face-center containment.
    This is intended for testing, not production tracking.
    """

    def __init__(self, min_containment=0.80):
        self.min_containment = min_containment

    def associate(self, faces, people):
        associations = []

        for face_index, face in enumerate(faces):
            fx1, fy1, fx2, fy2 = map(
                float,
                face.bbox,
            )

            face_area = max(
                0.0,
                (fx2 - fx1) * (fy2 - fy1),
            )

            if face_area <= 0:
                continue

            best_match = None
            best_overlap = 0.0

            for person_index, person in enumerate(people):
                px1, py1, px2, py2 = person["bbox"]

                ix1 = max(fx1, px1)
                iy1 = max(fy1, py1)
                ix2 = min(fx2, px2)
                iy2 = min(fy2, py2)

                intersection_area = max(
                    0.0,
                    ix2 - ix1,
                ) * max(
                    0.0,
                    iy2 - iy1,
                )

                containment = intersection_area / face_area

                if (
                    containment >= self.min_containment
                    and containment > best_overlap
                ):
                    best_overlap = containment
                    best_match = person_index

            associations.append({
                "face_index": face_index,
                "person_index": best_match,
                "containment": best_overlap,
            })

        return associations

from __future__ import annotations

import numpy as np


class BodyReID:
    """Small embedding-based body re-identification utility used for prototypes."""

    def __init__(self, dim=None):
        self.dim = dim

    @staticmethod
    def _as_vector(embedding):
        vector = np.asarray(embedding, dtype=np.float32).reshape(-1)
        if vector.size == 0:
            raise ValueError("Embedding is empty.")
        if not np.isfinite(vector).all():
            raise ValueError("Embedding contains non-finite values.")

        norm = np.linalg.norm(vector)
        if norm == 0:
            raise ValueError("Embedding norm is zero.")
        return vector / norm

    def extract_embedding(self, image, bbox=None):
        if image is None or np.asarray(image).size == 0:
            raise ValueError("Input image is empty or invalid.")

        arr = np.asarray(image, dtype=np.float32)
        if bbox is not None:
            x1, y1, x2, y2 = [int(v) for v in bbox]
            arr = arr[y1:y2, x1:x2]
            if arr.size == 0:
                raise ValueError("Body crop is empty.")

        vector = arr.reshape(-1)
        if self.dim is not None and vector.size != self.dim:
            vector = vector[: self.dim]

        return self._as_vector(vector)

    def cosine_similarity(self, a, b):
        a_vec = self._as_vector(a)
        b_vec = self._as_vector(b)
        return float(np.dot(a_vec, b_vec))

    def match(self, query_embedding, gallery, threshold=0.5):
        if not gallery:
            return {"identity": "Unknown", "similarity": None, "status": "no_candidates", "candidates": []}

        ranked = []
        for name, embeddings in gallery.items():
            if not embeddings:
                continue
            sims = [self.cosine_similarity(query_embedding, embedding) for embedding in embeddings]
            ranked.append({"name": name, "similarity": max(sims)})

        ranked.sort(key=lambda item: item["similarity"], reverse=True)
        best = ranked[0]
        if best["similarity"] < threshold:
            return {"identity": "Unknown", "similarity": best["similarity"], "status": "below_threshold", "candidates": ranked}
        return {"identity": best["name"], "similarity": best["similarity"], "status": "candidate_match", "candidates": ranked}

from dataclasses import dataclass, asdict

import cv2
import numpy as np


@dataclass
class FaceQualityResult:
    acceptable: bool
    reason: str

    width: int
    height: int

    blur_score: float
    detection_confidence: float

    size_ok: bool
    blur_ok: bool

class FaceQualityChecker:
    """
    Experimental face quality assessment.

    Thresholds are initial settings only. They must be validated
    against representative CCTV footage before production use.
    """

    def __init__(
        self,
        min_face_width=60,
        min_face_height=60,
        min_blur_score=30.0,
        min_detection_confidence=0.50,
    ):
        self.min_face_width = min_face_width
        self.min_face_height = min_face_height
        self.min_blur_score = min_blur_score
        self.min_detection_confidence = min_detection_confidence

    def evaluate(self, image, face):
        if image is None or image.size == 0:
            return FaceQualityResult(
                False,
                "invalid_image",
                0,
                0,
                0.0,
                0.0,
                False,
                False,
                False,
            )

        image_height, image_width = image.shape[:2]

        bbox = np.asarray(face.bbox, dtype=float)

        if bbox.shape != (4,) or not np.isfinite(bbox).all():
            return FaceQualityResult(
                False,
                "invalid_face_box",
                0,
                0,
                0.0,
                0.0,
                False,
                False,
                False,
            )

        x1, y1, x2, y2 = bbox.tolist()

        confidence = float(face.det_score)

        if (
            not np.isfinite(confidence)
            or x1 < 0
            or y1 < 0
            or x2 > image_width
            or y2 > image_height
            or x2 <= x1
            or y2 <= y1
        ):
            return FaceQualityResult(
                False,
                "invalid_or_outside_face_box",
                0,
                0,
                0.0,
                confidence if np.isfinite(confidence) else 0.0,
                False,
                False,
                False,
            )

        # Use floor/ceil to avoid accidentally shrinking the crop.
        x1, y1 = int(np.floor(x1)), int(np.floor(y1))
        x2, y2 = int(np.ceil(x2)), int(np.ceil(y2))

        width = x2 - x1
        height = y2 - y1

        face_crop = image[y1:y2, x1:x2]

        if face_crop.size == 0:
            return FaceQualityResult(
                False,
                "empty_face_crop",
                width,
                height,
                0.0,
                confidence,
                False,
                False,
                confidence >= self.min_detection_confidence,
            )

        gray = cv2.cvtColor(face_crop, cv2.COLOR_BGR2GRAY)

        blur_score = float(
            cv2.Laplacian(gray, cv2.CV_64F).var()
        )

        size_ok = (
            width >= self.min_face_width
            and height >= self.min_face_height
        )

        blur_ok = blur_score >= self.min_blur_score

        confidence_ok = (
            confidence >= self.min_detection_confidence
        )

        failed_checks = []

        if not size_ok:
            failed_checks.append("face_too_small")

        if not blur_ok:
            failed_checks.append("face_too_blurry")

        if not confidence_ok:
            failed_checks.append("low_detection_confidence")

        acceptable = not failed_checks

        reason = (
            "acceptable"
            if acceptable
            else ",".join(failed_checks)
        )

        return FaceQualityResult(
            acceptable=acceptable,
            reason=reason,
            width=width,
            height=height,
            blur_score=blur_score,
            detection_confidence=confidence,
            size_ok=size_ok,
            blur_ok=blur_ok,
            confidence_ok=confidence_ok,
        )

class FaceRecognizer:
    def __init__(self):
        self.app = insightface.app.FaceAnalysis(
            name="buffalo_l",
            providers=["CPUExecutionProvider"],
        )
        self.app.prepare(ctx_id=-1, det_size=(320 , 320))

    @staticmethod
    def normalize_embedding(embedding):
        embedding = np.asarray(embedding, dtype=np.float32)
        norm = np.linalg.norm(embedding)

        if not np.isfinite(norm) or norm == 0:
            raise ValueError("Invalid face embedding.")

        return embedding / norm

    def extract_embedding(self, image_path):
        image_path = Path(image_path)
        image = cv2.imread(str(image_path))

        if image is None:
            raise ValueError(f"Could not read image: {image_path}")

        faces = self.app.get(image)

        if len(faces) == 0:
            raise ValueError(f"No face detected: {image_path}")

        if len(faces) > 1:
            raise ValueError(
                f"Multiple faces detected in enrollment image: {image_path}"
            )

        return self.normalize_embedding(faces[0].normed_embedding)

    def load_gallery(self, gallery_path):
        gallery_path = Path(gallery_path)
        gallery = {}

        for staff_dir in sorted(gallery_path.iterdir()):
            if not staff_dir.is_dir():
                continue

            face_dir = staff_dir / "face"

            if not face_dir.is_dir():
                continue

            embeddings = []

            for image_path in sorted(face_dir.iterdir()):
                if image_path.suffix.lower() not in {
                    ".jpg", ".jpeg", ".png", ".bmp"
                }:
                    continue

                try:
                    embedding = self.extract_embedding(image_path)
                    embeddings.append(embedding)
                    print(f"Enrolled image: {staff_dir.name}/{image_path.name}")

                except ValueError as exc:
                    print(f"Skipped image: {exc}")

            if embeddings:
                gallery[staff_dir.name] = embeddings

        if not gallery:
            raise RuntimeError("No valid staff embeddings were enrolled.")

        return gallery

    def rank_candidates(self, query_embedding, gallery):
        query_embedding = self.normalize_embedding(query_embedding)
        results = []

        for staff_name, reference_embeddings in gallery.items():
            # Use the highest reference-image similarity for this
            # initial prototype. We will evaluate this strategy later.
            similarities = [
                float(np.dot(query_embedding, reference))
                for reference in reference_embeddings
            ]

            results.append({
                "name": staff_name,
                "similarity": max(similarities),
            })

        return sorted(
            results,
            key=lambda result: result["similarity"],
            reverse=True,
        )

    def identify(self, query_embedding, gallery, threshold=0.30):
        """
        Demonstration only.
        The threshold has not been validated for production use.
        """
        ranked = self.rank_candidates(query_embedding, gallery)

        if not ranked:
            return {
                "identity": "Unknown",
                "similarity": None,
                "status": "no_candidates",
                "candidates": [],
            }

        best = ranked[0]

        if best["similarity"] < threshold:
            return {
                "identity": "Unknown",
                "similarity": best["similarity"],
                "status": "below_threshold",
                "candidates": ranked,
            }

        return {
            "identity": best["name"],
            "similarity": best["similarity"],
            "status": "candidate_match",
            "candidates": ranked,
        }

from __future__ import annotations


class StaffMatcher:
    """Utility for ranking gallery identities by similarity in a simple prototype workflow."""

    def __init__(self, threshold=0.30):
        self.threshold = threshold

    def match(self, query_embedding, gallery):
        if not gallery:
            return {"identity": "Unknown", "similarity": None, "status": "no_candidates", "candidates": []}

        ranked = []
        for staff_name, embeddings in gallery.items():
            if not embeddings:
                continue
            similarities = [float(similarity) for similarity in embeddings]
            best_similarity = max(similarities, default=0.0)
            ranked.append({"name": staff_name, "similarity": best_similarity})

        ranked.sort(key=lambda result: result["similarity"], reverse=True)
        if not ranked:
            return {"identity": "Unknown", "similarity": None, "status": "no_candidates", "candidates": []}

        best = ranked[0]
        if best["similarity"] < self.threshold:
            return {"identity": "Unknown", "similarity": best["similarity"], "status": "below_threshold", "candidates": ranked}
        return {"identity": best["name"], "similarity": best["similarity"], "status": "candidate_match", "candidates": ranked}

from __future__ import annotations


class CrossCameraTracker:
    """Prototype cross-camera tracker that groups all detections by identity label."""

    def __init__(self):
        self.identities = {}

    def update(self, detections):
        for detection in detections or []:
            identity = detection.get("identity", "Unknown") if isinstance(detection, dict) else detection
            self.identities.setdefault(identity, []).append(detection)
        return dict(self.identities)

    def clear(self):
        self.identities.clear()

from __future__ import annotations


class PersonTracker:
    """Prototype tracker that keeps track of recent detections by ID."""

    def __init__(self):
        self.tracks = {}

    def update(self, detections):
        self.tracks = {idx: detection for idx, detection in enumerate(detections or [])}
        return list(self.tracks.values())

    def clear(self):
        self.tracks.clear()

from pathlib import Path
import sys

import cv2

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from src.detection.person_detector import PersonDetector


SUPPORTED_EXTENSIONS = {
    ".jpg", ".jpeg", ".png", ".bmp"
}

EVALUATION_PATH = ROOT / "data" / "evaluation"
OUTPUT_PATH = ROOT / "tests" / "person_detection_output"


def main():
    OUTPUT_PATH.mkdir(parents=True, exist_ok=True)

    detector = PersonDetector(
        model_path="yolo11n.pt",
        confidence=0.35,
        image_size=640,
    )

    print("YOLO person detection test")
    print("=" * 60)

    tested = 0

    for person_dir in sorted(EVALUATION_PATH.iterdir()):
        if not person_dir.is_dir():
            continue

        for image_path in sorted(person_dir.iterdir()):
            if image_path.suffix.lower() not in SUPPORTED_EXTENSIONS:
                continue

            image = cv2.imread(str(image_path))

            if image is None:
                print(f"Skipped unreadable image: {image_path}")
                continue

            detections = detector.detect(image)

            output_image = detector.draw_detections(
                image,
                detections,
            )

            output_name = (
                f"{person_dir.name}_{image_path.stem}_persons.jpg"
            )

            output_path = OUTPUT_PATH / output_name

            if not cv2.imwrite(str(output_path), output_image):
                print(f"Could not save: {output_path}")
                continue

            print(f"\nImage: {person_dir.name}/{image_path.name}")
            print(f"People detected: {len(detections)}")

            for index, detection in enumerate(detections, start=1):
                print(
                    f"  Person {index}: "
                    f"bbox={detection['bbox']}, "
                    f"confidence={detection['confidence']:.3f}"
                )

            print(f"Saved visualization: {output_path}")
            tested += 1

    print(f"\nImages processed successfully: {tested}")
    print(f"Output directory: {OUTPUT_PATH}")


if __name__ == "__main__":
    main()
